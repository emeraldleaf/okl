```csharp
app.MapGet("/orders/search", (HttpContext ctx, string? q) =>
{
    var user = Auth.CurrentUser(ctx);
    if (user is null) return Results.Unauthorized();
    var term = q ?? "";
    return Results.Ok(Store.Orders.Values
        .Where(o => o.Owner == user && o.Item.Contains(term, StringComparison.OrdinalIgnoreCase)));
});
```
